def cadastrar_issues_em_lote(repo: str, token: str, backlog: list[dict]) -> None:
    """Cadastra uma lista de issues via API REST v3 do GitHub garantindo tratamento de erros e throttling."""
    
    if token == "COLE_SEU_TOKEN_AQUI" or not token:
        raise ValueError("Token de autenticação do GitHub não foi fornecido.")

    url = f"https://api.github.com/repos/{repo}/issues"
    
    # Session reutiliza a conexão TCP/TLS
    session = requests.Session()
    session.headers.update({
        "Authorization": f"Bearer {token}",
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28"
    })

    print(f"🚀 Iniciando importação de {len(backlog)} issues para '{repo}'...\n")

    for index, issue_payload in enumerate(backlog, start=1):
        try:
            response = session.post(url, json=issue_payload, timeout=(3.05, 10))
            response.raise_for_status()
            
            data = response.json()
            print(f"✅ [{index}/{len(backlog)}] Issue #{data['number']} criada: {data['html_url']}")

            # Throttling preventivo contra Rate Limit da API
            time.sleep(1)

        except requests.exceptions.HTTPError as err:
            status_code = err.response.status_code
            if status_code in (401, 403):
                print(f"❌ [ERRO {status_code}] Falha de autorização ou Rate Limit excedido ao tentar criar '{issue_payload['title']}'.")
                print(f"Detalhes: {err.response.text}")
                break
            else:
                print(f"⚠️ [ERRO {status_code}] Falha ao criar issue '{issue_payload['title']}': {err.response.text}")

        except requests.exceptions.RequestException as err:
            print(f"💥 [ERRO DE REDE] Exceção de I/O ao processar '{issue_payload['title']}': {err}")

if __name__ == "__main__":
    cadastrar_issues_em_lote(REPO_PATH, GITHUB_TOKEN, ISSUES_BACKLOG)
